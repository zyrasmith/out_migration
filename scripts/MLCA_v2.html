x <- data |> 
  mutate(
         public = case_when(
            control == 2 ~ 0,                             # 0: private
            control == 1 ~ 1,                             # 1: public 
            control == 3 ~ 2,                             # 2: for-profit
            TRUE ~ NA_real_),   
         
         twoyear = case_when(
            iclevel == 1 ~ 0,                             # 0: 4 or more
            iclevel == 2 ~ 1,                             # 1: at least 2 & less than 4
            TRUE ~ NA_real_),     
         
         # msi = case_when(                                 # 0: non of them
         #    hbcu == 1 ~ 1,                                # 1: HBCU
         #    hsi == 1 ~ 2,                                 # 2: HSI
         #    tribal == 1 ~ 3,                              # 3: TCU
         #    hbcu == 2 & hsi == 0 & tribal == 2 ~ 0,
         #    is.na(hbcu) | is.na(hsi) | is.na(tribal) ~ NA_real_,
         #    TRUE ~ NA_real_),
         
         msi = case_when(
           hbcu == 1 | hsi == 1 | tribal == 1 ~ 1, 
           TRUE ~ 0),
         
         rurality = case_when(
            locale %in% c(31,32,33,41,42,43) ~ 0,         # 0: Town + Rural
            locale %in% c(21,22,23) ~ 1,                  # 1: Suburb
            locale %in% c(11,12,13) ~ 2,                  # 2: City
            TRUE ~ NA_real_),        
         
         size = case_when(
           instsize == 1 ~ 0,                             # 0: < 1,000
           instsize == 2 ~ 1,                             # 1: 1,000 - 4,999
           instsize == 3 ~ 2,                             # 1: 5,000 - 9,999
           instsize == 4 ~ 3,                             # 2: 10,000 - 19,999
           instsize == 5 ~ 4,                             # 3: >=20,000
           TRUE ~ NA_real_),
         
         tuition = case_when(
           tuition1 == 0 & obereg != 0 ~ NA_real_,
           tuition1 == 0 & obereg == 0 ~ 0,
           tuition1 < 10000 ~ 0,                           # 0: < $10,000
           tuition1 >= 10000 & tuition1 < 20000 ~ 1,       # 1: $10,000 - $19,999
          # tuition1 >= 20000 & tuition1 < 30000 ~ 2,       # 2: $20,000 - $29,999
           tuition1 >= 20000 & tuition1 < 40000 ~ 2,       # 3: $30,000 - $39,999
          # tuition1 >= 40000 & tuition1 < 50000 ~ 4,       # 4: $40,000 - $49,999
           tuition1 >= 40000                    ~ 3,       # 5: > $50,000
           TRUE ~ NA_real_),
         

         housing = case_when(
           room == 2 ~ 0,                                 # 0: no housing
           room == 1 ~ 1,                                 # 1: offer housing
           TRUE ~ NA_real_),    

         religious = ifelse(relaffil == -2, 0, 1),        # 0: no religious
                                                          # 1: any religious
         
        #We need to remove colleges that offer completely distance education from our sample
          distance = case_when(
           distnced == 2 ~ 0,                             # 0: not completely distanced learning
           distnced == 1 ~ 1,                             # 1: completely distanced learning
           TRUE ~NA_real_),    

         sports = case_when(
           athassoc == 2 ~ 0,                             # 0: not a member
           athassoc == 1 ~ 1,                             # 1: member of National Athletic Association
           TRUE ~NA_real_),    

         loan = case_when(
           is.na(loan_a) ~ NA_real_,
           loan_a <= quantile(loan_a, 0.25, na.rm = TRUE) ~ 0,     # 0: lowest
           loan_a <= quantile(loan_a, 0.50, na.rm = TRUE) ~ 1,     # 1: low 
           loan_a <= quantile(loan_a, 0.75, na.rm = TRUE) ~ 2,     # 2: high
           TRUE ~ 3),                                              # 3: highest
 
# Q. Since MLCA requires variables starting at 0, is it appropriate to divide loan amount into quartiles (0–3)?

         asc = case_when(
           carnegiesaec == 2 ~ 0,                         # 0: Higher Access, Lower Earnings 
           carnegiesaec == 4 ~ 1,                         # 1: Higher Access, Medium Earnings
           carnegiesaec == 6 ~ 2,                         # 2: Higher Access, Higher Earnings
           carnegiesaec == 1 ~ 3,                         # 3: Lower Access, Lower Earnings
           carnegiesaec == 3 ~ 4,                         # 4: Lower Access, Medium Earnings
           carnegiesaec == 5 ~ 5,                         # 5: Lower Access, Higher Earnings
           TRUE ~ NA_real_),

         awardlev = case_when(
           carnegiealf %in% c(1,2) ~ 0,                    # 0: associates 
           carnegiealf == 3 ~ 1,                            # 1: baccalaureate
           carnegiealf == 4 ~ 2,                           # 2: undergrad / grad-Master's
           carnegiealf == 5 ~ 3,                           # 3: undergrad / grad-Doctorate 
           
           is.na(carnegiealf) & (hloffer == 3 | hloffer == 4) ~ 0,
           carnegiealf == -2 & (hloffer == 3 | hloffer == 4) ~ 0,
           hloffer == 5 & instcat == 2 ~ 1,
           
           TRUE ~ NA_real_),
